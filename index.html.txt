<script>
// ── CONFIG — ndërroni këtë me URL-në tuaj të Render ──
const API = 'https://asistent-zyre-backend.onrender.com';

let token = localStorage.getItem('az_token') || '';
let history = [], selectedDocs = new Set(), allDocs = [];
let busy = false;

// ── AUTH ──
async function doLogin() {
  const u = document.getElementById('l-user').value.trim();
  const p = document.getElementById('l-pass').value;
  document.getElementById('l-err').textContent = '';
  try {
    const r = await fetch(API + '/api/login', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username: u, password: p })
    });
    const d = await r.json();
    if (!r.ok) { document.getElementById('l-err').textContent = d.error || 'Gabim'; return; }
    token = d.token;
    localStorage.setItem('az_token', token);
    showApp();
  } catch { document.getElementById('l-err').textContent = 'Nuk mund të lidhem me serverin.'; }
}

function logout() {
  localStorage.removeItem('az_token');
  token = '';
  document.getElementById('app').style.display = 'none';
  document.getElementById('login-screen').style.display = 'flex';
}

function authHdr() { return { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' }; }

async function showApp() {
  document.getElementById('login-screen').style.display = 'none';
  document.getElementById('app').style.display = 'flex';
  await loadDocs();
}

if (token) showApp();

// ── DOCUMENTS ──
async function loadDocs() {
  try {
    const r = await fetch(API + '/api/documents', { headers: authHdr() });
    if (r.status === 401) { logout(); return; }
    allDocs = await r.json();
    renderFiles();
  } catch (e) { console.error(e); }
}

function renderFiles() {
  const list = document.getElementById('file-list');
  list.innerHTML = '';
  allDocs.forEach(d => {
    const sel = selectedDocs.has(d.id);
    const div = document.createElement('div');
    div.className = 'fi' + (sel ? ' selected' : '');
    div.onclick = () => toggleDoc(d.id);
    div.innerHTML = `
      <span class="ficon">${fIcon(d.file_type)}</span>
      <div style="flex:1;min-width:0">
        <div class="fname">${d.name}</div>
        <div class="fmeta">${(d.size_chars||0).toLocaleString()} kar · ${fDate(d.created_at)}</div>
      </div>
      <button class="fdel" onclick="event.stopPropagation();delDoc('${d.id}')" title="Fshi">✕</button>`;
    list.appendChild(div);
  });
  document.getElementById('doc-count').textContent = allDocs.length + ' skedarë';
}

function toggleDoc(id) {
  if (selectedDocs.has(id)) selectedDocs.delete(id);
  else selectedDocs.add(id);
  renderFiles();
}

async function delDoc(id) {
  if (!confirm('Fshi këtë dokument nga baza?')) return;
  await fetch(API + '/api/documents/' + id, { method: 'DELETE', headers: authHdr() });
  allDocs = allDocs.filter(d => d.id !== id);
  selectedDocs.delete(id);
  renderFiles();
}

function fIcon(t) {
  if (t === 'pdf') return '📄';
  if (['doc','docx'].includes(t)) return '📝';
  if (['xls','xlsx'].includes(t)) return '📊';
  if (['ppt','pptx'].includes(t)) return '📋';
  return '🖼️';
}
function fDate(s) { return s ? new Date(s).toLocaleDateString('sq-AL') : ''; }

// ── UPLOAD ──
async function handleFiles(files) {
  for (const file of files) {
    const idx = allDocs.length;
    // Show uploading placeholder
    allDocs.unshift({ id: '_up_'+idx, name: file.name, file_type: file.name.split('.').pop(), size_chars: 0, created_at: new Date().toISOString(), _uploading: true });
    renderFiles();

    const fd = new FormData();
    fd.append('file', file);
    try {
      const r = await fetch(API + '/api/upload', {
        method: 'POST',
        headers: { 'Authorization': 'Bearer ' + token },
        body: fd
      });
      const d = await r.json();
      // Replace placeholder with real doc
      const pi = allDocs.findIndex(x => x.id === '_up_'+idx);
      if (pi >= 0 && d.id) allDocs[pi] = { id: d.id, name: d.name, file_type: file.name.split('.').pop(), size_chars: d.size_chars, created_at: new Date().toISOString() };
    } catch (e) {
      allDocs = allDocs.filter(x => x.id !== '_up_'+idx);
    }
    renderFiles();
  }
  document.getElementById('fi').value = '';
}

// ── CHAT ──
function ar(el) { el.style.height = 'auto'; el.style.height = Math.min(el.scrollHeight, 120) + 'px'; }
function hk(e) { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMsg(); } }
function sq(t) { const el = document.getElementById('user-input'); el.value = t; el.focus(); ar(el); }

function addMsg(role, txt) {
  document.getElementById('welcome-msg')?.remove();
  const d = document.createElement('div');
  d.className = 'msg ' + role;
  d.innerHTML = `<div class="av">${role === 'assistant' ? 'AI' : 'Ju'}</div><div class="bubble">${esc(txt)}</div>`;
  document.getElementById('chat-area').appendChild(d);
  document.getElementById('chat-area').scrollTop = 99999;
}
function showTyp() {
  document.getElementById('welcome-msg')?.remove();
  const d = document.createElement('div'); d.className = 'msg assistant'; d.id = 'typ';
  d.innerHTML = `<div class="av">AI</div><div class="bubble"><div class="typing"><span></span><span></span><span></span></div></div>`;
  document.getElementById('chat-area').appendChild(d);
  document.getElementById('chat-area').scrollTop = 99999;
}
function remTyp() { document.getElementById('typ')?.remove(); }
function esc(t) { return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/\n/g,'<br>'); }
function clearChat() { history = []; document.getElementById('chat-area').innerHTML = '<div class="welcome" id="welcome-msg"><strong>Biseda u pastrua.</strong><br/>Gati për fillim të ri.</div>'; }

async function sendMsg() {
  const el = document.getElementById('user-input');
  const text = el.value.trim();
  if (!text || busy) return;
  busy = true;
  document.getElementById('send-btn').disabled = true;
  el.value = ''; el.style.height = 'auto';
  addMsg('user', text);
  showTyp();

  try {
    const r = await fetch(API + '/api/chat', {
      method: 'POST', headers: authHdr(),
      body: JSON.stringify({
        message: text,
        history: history.slice(-10),
        docIds: selectedDocs.size > 0 ? [...selectedDocs] : null
      })
    });
    const d = await r.json();
    remTyp();
    const reply = d.reply || 'Gabim.';
    addMsg('assistant', reply);
    history.push({ role: 'user', content: text }, { role: 'assistant', content: reply });
  } catch {
    remTyp();
    addMsg('assistant', 'Gabim lidhjeje me serverin. Provoni përsëri.');
  }
  busy = false;
  document.getElementById('send-btn').disabled = false;
}
